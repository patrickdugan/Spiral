  stockB/.style={line width=2.6pt, black!30},
  plain/.style={line width=1.1pt, black!35},

% (a) per-agent channels
\begin{scope}

  \draw[plain] (h) -- (1.5,3.2);

  \node[net] at (1.5,3.45) {rest of network};
  \foreach \x in {0,1,2,3} \node[vtx] at (a\x) {$a_\x$};
  \node[vtx] at (h) {$h$};
  \node[lab, align=center] at (1.5,-0.95) {(a) channels $C$\\ $k$ stocks, one per edge: $\sum_a\hat f_a$};

\begin{scope}[xshift=5.1cm]
  \coordinate (S) at (1.5,2.1);

  \draw[stockB] (-0.35,1.18) -- (3.35,1.18);
  \node[lab, below, fill=black!7, inner sep=1.5pt] at (1.5,1.08) {$\hat f_S$: one stock for every holder};
  \node[draw, black!60, rounded corners=2pt, fill=white, lab, align=center, inner sep=2pt] at (-0.55,2.1) {$W_S$ fronted,\\ unswept};
  \node[vtx] at (S) {$S$};
  \node[lab, align=center] at (1.5,-0.95) {(b) virtual outputs $V$\\ one stock, $\hat f_S$, plus $W_S$};

\caption{Where the forecast lives, for the same $k$ agents and a hub. Light segments are stock placed by forecast before demand is known; dark segments are the agents' own balances. (a) Per-agent channels: the hub holds a separate stock on every edge, and the total is a sum of $k$ forecast peaks. (b) An Ark: outputs have no direction at the holder tier, the server co-signs every transfer, its stock is forecast once for all holders, and value fronted at spends and refreshes waits unswept as $W_S$ (Proposition~\ref{prop:relocation}). (c) A multiparty channel: one shared output whose hub part is the single stock for the group, forecast once, with nothing fronted and every update signed by all $k+1$ members (Proposition~\ref{prop:hyperpool}). The channel from $h$ or $S$ to the rest of the network carries a tier common to the three objects and is drawn without a stock.}
\label{fig:objects}
\end{figure}
